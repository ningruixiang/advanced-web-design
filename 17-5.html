常见坑 & 速查表
定时器最容易踩的 4 个坑，以及对应的 clearTimeout / clearInterval 清理方式。
// ❌ 坑 1：setInterval 叠加 —— 点两次 start 会开两个！
let timer = setInterval(fn, 1000);
// ✅ 解决：每次新开之前先 clear 掉旧的
clearInterval(timer); timer = setInterval(fn, 1000);

// ❌ 坑 2：interval 不随页面 / 组件销毁自动停
window.onbeforeunload = () => clearInterval(timer);  // 主动清理

// ❌ 坑 3：interval 不关心上一个回调有没有执行完（长任务叠加）
function longJob() { return new Promise(r => setTimeout(r, 2000)); }
// ✅ 解决：递归 setTimeout 实现"串行"间隔
let running = false;
function loop() {
  if (running) return;
  running = true;
  longJob().finally(() => { running = false; setTimeout(loop, 1000); });
}

// ❌ 坑 4：this 绑定 / 闭包 —— 箭头函数救了半条命
setTimeout(() => this.update(), 1000);  // ✅ 箭头函数继承外层 this
setTimeout(this.update.bind(this), 1000);     // ✅ 显式 bind
