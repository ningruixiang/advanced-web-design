fetch 标准写法
下面就是这次实际跑的代码模板。fetch 返回 Promise，用 .then() 或 async/await 都行。
fetch("https://api.github.com/zen")        // ① 发起请求，返回 Promise
  .then(res => {
    if (!res.ok) throw new Error("HTTP " + res.status);  // ② 检查状态码
    return res.json();                 // ③ 把响应体解析成 JSON
  })
  .then(data => {
    console.log(data);                       // ④ 拿到真正的数据对象
  })
  .catch(err => {
    console.error(err);                      // ⑤ 网络错误 / 解析失败
  });
